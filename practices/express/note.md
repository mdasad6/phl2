// Node Project Setup Steps : ---------------------------------------------

01. npm init -y                           // package.json come

02. npm i -D typescript                 // typescript compiler install (dev-dependency)

03. npx tsc --init                    // tsconfig.json come

04. npm i express                   // express install 

05. npm i --save-dev @types/express 

06. npm i -D tsx                 // server run

07. npm i pg                   // Non-blocking PostgreSQL client for Node.jsI      

08. npm i --save-dev @types/pg

09. npm i dotenv            // 

10. npm i bcryptjs        //

11. npm i jsonwebtoken  //

12. npm i --save-dev @types/jsonwebtoken
